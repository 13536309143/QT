55 · QVulkanWindow 完整清屏：回调、命令与 Qt 的责任

一、窗口与渲染器为什么分开

ClearWindow 继承 QVulkanWindow，只负责提供 createRenderer；ClearRenderer 继承 QVulkanWindowRenderer，处理 GPU 相关回调。createRenderer 返回的渲染器由窗口管理，不应在外部又独立 delete。渲染器保留窗口观察指针，是因为窗口寿命覆盖它的回调使用期。

本例使用单采样默认 render pass：颜色与深度/模板两个附件。因此 clear 数组是两个元素，setSampleCount(1) 明确保持这一前提。以后开启 MSAA，需要重新核对默认 pass 的附件结构，不能原样沿用所有计数。

二、回调与资源放置表

回调 什么阶段调用 适合放什么 不应做什么
preInitResources 设备初始化前的准备阶段 需要尽早完成的配置 访问尚不存在的 device
initResources 设备可用后 设备级资源与函数入口 永久缓存本次 framebuffer
initSwapChainResources 交换链建立或重建后 尺寸相关资源、相容管线 假定只调用一次
startNextFrame Qt 提供本帧录制机会 记录当前帧命令 再自己提交 Qt 管理的 command
releaseSwapChainResources 旧交换链资源释放前 释放自建尺寸相关资源 销毁 Qt 的默认交换链对象
releaseResources 设备相关资源释放阶段 释放自建 device 对象 继续请求使用旧 device

回调名称是 Qt 的生命周期契约，不能仅凭 C++ 构造/析构推断所有 Vulkan 资源都安全。设备丢失还有独立处理接口；复杂程序应停止使用失效资源并进入恢复或退出路径。

三、initResources 中取得函数入口

instance->deviceFunctions(window->device()) 返回该逻辑设备的 QVulkanDeviceFunctions 指针。通过它调用 vkCmdBeginRenderPass 等设备级接口。这不是创建第二个 device，也不是给函数添加自动同步；它只是 Qt 提供的设备入口访问方式。

本例没有直接调用全局 vkCreateInstance，所以不需要为这些包装调用额外手拼 Vulkan 导入库。若后续改为直接加载或调用原生入口，应重新明确链接和函数装载策略，不能把两种风格不加区分地混用。

四、逐行拆解 startNextFrame

VkClearValue 是联合体：颜色附件填 color，深度附件填 depthStencil。颜色四个分量设置为 0.10、0.32、0.48、1.0，深度清为 1、模板为 0。具体清值是否使用取决于 render pass 的附件 loadOp，清值不是无条件覆盖任意图像的独立命令。

VkRenderPassBeginInfo 先零初始化，再设置 sType。renderPass 来自 defaultRenderPass，framebuffer 来自 currentFramebuffer，extent 来自 swapChainImageSize。这里需要实际像素尺寸，不能直接使用逻辑窗口宽高。

currentCommandBuffer 是 Qt 为当前帧提供、已进入适当录制阶段的命令缓冲。本例在其中 beginRenderPass，再 endRenderPass。清屏不需要绑定三角形 pipeline，所以本章还没有 vertex shader 或 vkCmdDraw。

最后调用 frameReady，告诉 Qt 本帧录制准备好了。Qt 负责其管理路径上的队列提交、呈现及相关帧调度；应用不要对同一 command 再 vkEndCommandBuffer 或 vkQueueSubmit。frameReady 不是阻塞等待 GPU 完成的函数。

五、静态内容和动画怎样继续

静态清屏只需当前帧。持续动画可在结束时 requestUpdate 请求下一帧，但不能忘记 frameReady，也不应该无限递归调用 startNextFrame。需要耗时 CPU 准备时，把普通数据准备与 GPU 命令录制分开，保持对象线程约束。

缩放会触发交换链重建，currentFramebuffer 和 imageSize 都可能改变，因此每帧重新取得当前值。窗口最小化时可能没有可呈现尺寸，不在任意定时器里强行构建零尺寸图像。

六、动手与验收

运行 18_vulkan，改动颜色，重新编译确认变化；反复缩放和最小化恢复。打开 --validation 需要先安装该层；记录第一条消息及对象类型，先消除错误再加入三角形。

验收：能指出哪个对象由 Qt 拥有，哪个指针是观察者；能解释当前帧、当前交换链图像和用户业务任务是不同概念。本例通过帧计数确认录制回调执行，颜色和实际呈现仍应人工观察。

配套完整源码

在根工程构建本章目标：18_vulkan。Vulkan 需先启用可选构建。

main.cpp

#include <QGuiApplication>
#include <QVulkanInstance>
#include <QVulkanWindow>
#include <QVulkanFunctions>
#include <QTimer>
#include <QDebug>

class ClearRenderer final : public QVulkanWindowRenderer {
public:
    explicit ClearRenderer(QVulkanWindow *window, int *frames) : window_(window), frames_(frames) {}
    void initResources() override {
        functions_ = window_->vulkanInstance()->deviceFunctions(window_->device());
    }
    void startNextFrame() override {
        VkClearValue clear[2] = {};
        clear[0].color.float32[0] = 0.10f;
        clear[0].color.float32[1] = 0.32f;
        clear[0].color.float32[2] = 0.48f;
        clear[0].color.float32[3] = 1.0f;
        clear[1].depthStencil = {1.0f, 0};
        VkRenderPassBeginInfo begin = {};
        begin.sType = VK_STRUCTURE_TYPE_RENDER_PASS_BEGIN_INFO;
        begin.renderPass = window_->defaultRenderPass();
        begin.framebuffer = window_->currentFramebuffer();
        begin.renderArea.extent.width = uint32_t(window_->swapChainImageSize().width());
        begin.renderArea.extent.height = uint32_t(window_->swapChainImageSize().height());
        begin.clearValueCount = 2;  // 本例固定单采样：颜色 + 深度/模板。
        begin.pClearValues = clear;
        const auto command = window_->currentCommandBuffer();
        functions_->vkCmdBeginRenderPass(command, &begin, VK_SUBPASS_CONTENTS_INLINE);
        functions_->vkCmdEndRenderPass(command);
        ++*frames_;
        window_->frameReady();
        // 静态清屏只需一帧；动画可在这里 requestUpdate()。
    }
private:
    QVulkanWindow *window_;
    QVulkanDeviceFunctions *functions_ = nullptr;
    int *frames_;
};
class ClearWindow final : public QVulkanWindow {
public:
    explicit ClearWindow(int *frames) : frames_(frames) {}
    QVulkanWindowRenderer *createRenderer() override { return new ClearRenderer(this, frames_); }
private:
    int *frames_;
};
int main(int argc, char **argv) {
    QGuiApplication app(argc, argv);
    QVulkanInstance instance;
    instance.setApiVersion(QVersionNumber(1, 0, 0));
    if (app.arguments().contains("--validation")) {
        if (!instance.supportedLayers().contains("VK_LAYER_KHRONOS_validation")) {
            qCritical() << "Requested validation layer is unavailable"; return 2;
        }
        instance.setLayers({"VK_LAYER_KHRONOS_validation"});
    }
    if (!instance.create()) { qCritical() << "Vulkan instance failed:" << instance.errorCode(); return 1; }
    int frames = 0;
    ClearWindow window(&frames); // 先销毁窗口，再销毁 instance 和帧计数。
    window.setVulkanInstance(&instance);
    window.setSampleCount(1);
    window.setTitle(QStringLiteral("示例 18 · Vulkan 清屏"));
    window.resize(640,480); window.show();
    if (app.arguments().contains("--self-test")) {
        QTimer::singleShot(2000, &window, [&] {
            if (!window.isValid() || frames == 0) { qCritical() << "No Vulkan frame recorded"; app.exit(3); }
            else { qInfo() << "PASS Vulkan instance, device, swapchain and command recording"; app.exit(0); }
        });
    }
    return app.exec();
}
