50 · Qt 与 OpenGL：上下文、控件及完整生命周期

前置：第 08、26、49 章。目标：运行 17_opengl，理解初始化、绘制和清理分别发生在哪个阶段。

一、选择正确的入口类

QOpenGLWidget 是 QWidget，可以加入布局，与按钮、表格一起使用；它把内容渲染到内部 framebuffer，再参与 Widgets 合成。QOpenGLWindow 是 QWindow 路线，适合独立渲染窗口；把它放到 Widgets 需要容器并理解原生窗口约束。初学时不要同时混合两种生命周期。

本例继承 QOpenGLWidget，同时 protected 继承 QOpenGLFunctions_3_3_Core。前者提供窗口与回调，后者提供 OpenGL 3.3 Core 函数入口。函数包装器并不自己创建上下文，也不保证当前机器有相应显卡能力。

二、格式必须先于上下文创建

在 QApplication 之前创建 QSurfaceFormat，调用 setVersion(3,3)、setProfile(CoreProfile)，最后 setDefaultFormat。它提出默认请求，不是把旧驱动升级成 3.3；初始化后仍应查看 context 的实际 format。请求失败要反馈，不能继续调用不可用的函数入口。

函数 参数与返回值 实际影响 易错点
setRenderableType(OpenGL) 枚举，返回 void 请求桌面 OpenGL 路线 与 OpenGL ES GLSL 语法不能随意混用
setVersion(3,3) 主次版本 int 请求所需能力 请求不等于支持
setProfile(CoreProfile) profile 枚举 选择现代核心配置 不再依赖旧固定管线
setDepthBufferSize(24) 位数 int 请求深度附件 请求深度不等于已启用 GL_DEPTH_TEST
setDefaultFormat(format) 格式对象 影响之后创建的上下文 控件已创建后再改通常太晚

三、Qt 在什么时机调用你

构造函数只创建普通 C++ 状态。控件首次需要显示时，Qt 建立渲染环境，再调用 initializeGL；窗口大小变化时调用 resizeGL；需要重绘时调用 paintGL。不要在构造函数直接创建 GPU buffer，因为那时没有保证可用的当前上下文。

initializeGL 中先 initializeOpenGLFunctions，检查 bool，再编译 program、建立 VAO/VBO。它通常不是每帧运行，但上下文被重建时可能重新进入，所以不能把“一生只初始化一次”当作所有平台的保证。

resizeGL 适合更新投影等尺寸相关状态。QOpenGLWidget 已管理其默认 framebuffer 和基本 viewport；手动离屏渲染后恢复自己的 viewport 时，应按实际目标像素尺寸设置，而不是盲目用逻辑 width/height。

paintGL 只做本帧工作：清除、绑定已建立资源、更新必要 uniform、绘制。不要每帧重新编译 shader、重新创建全部 buffer 或阻塞等待网络。

四、请求重绘与立即执行不同

普通槽函数里改变 angle_ 后调用 update()，让 Qt 稍后调度 paintGL。不要把鼠标事件当作有当前 GL 上下文的地方直接 glDrawArrays，也不要自己在 paintGL 末尾递归调用 paintGL。

makeCurrent() 在需要的回调之外把控件上下文设为当前；doneCurrent() 释放当前绑定。二者不等于创建或销毁上下文。若程序涉及多个控件或线程，不能认为“之前某处设过当前”现在还有效。

五、资源清理要早于基类销毁

本例析构函数显式 cleanup，并连接 context 的 aboutToBeDestroyed。cleanup 先 makeCurrent，再 destroy VAO、VBO 和 program，最后 doneCurrent。之所以不只依赖 C++ 成员析构，是因为成员销毁时必须仍有正确 GL 环境。

清理函数应允许多次调用；资源 handle 已清空时不重复使用。上下文意外丢失时，旧 GPU handle 不能当作新上下文资源继续画，应按重建路径重新建立。复杂应用还要处理共享上下文与线程归属，本例保持 GUI 单线程。

六、运行与定位失败

执行根目录 build.cmd，随后 scripts\run.cmd 17_opengl。自检用 scripts\run.cmd 17_opengl --self-test，它检查初始化、shader/buffer 建立，并读取中心和角落像素确认不是纯背景。需要正常 Windows 图形会话，不放入 offscreen 启动检查中假装测过 GPU。

无法建立上下文先检查驱动和实际版本；shader 失败读 program.log；纯背景再检查第 51 章的属性布局。不要把复制 Qt DLL 或反复调高版本号当作通用解决方案。

示例源码

配套完整源码

在根工程构建本章目标：17_opengl。Vulkan 需先启用可选构建。

main.cpp

#include <QtWidgets>
#include <QOpenGLWidget>
#include <QOpenGLFunctions_3_3_Core>
#include <QOpenGLShaderProgram>
#include <QOpenGLBuffer>
#include <QOpenGLVertexArrayObject>
#include <QSurfaceFormat>
#include <memory>

class Triangle final : public QOpenGLWidget, protected QOpenGLFunctions_3_3_Core {
public:
    explicit Triangle(QWidget *parent = nullptr) : QOpenGLWidget(parent) {}
    ~Triangle() override { QObject::disconnect(cleanupConnection_); cleanup(); }
    bool ready() const { return ready_; }
    QString error() const { return error_; }
protected:
    void initializeGL() override {
        functionsReady_ = initializeOpenGLFunctions();
        if (!functionsReady_) { error_ = "OpenGL 3.3 Core unavailable"; return; }
        cleanupConnection_ = connect(context(), &QOpenGLContext::aboutToBeDestroyed, this, [this] { cleanup(); }, Qt::DirectConnection);
        program_ = std::make_unique<QOpenGLShaderProgram>();
        const char *vertex = R"(#version 330 core
layout(location=0) in vec2 position;
layout(location=1) in vec3 color;
out vec3 vertexColor;
void main() { gl_Position=vec4(position,0.0,1.0); vertexColor=color; }
)";
        const char *fragment = R"(#version 330 core
in vec3 vertexColor;
out vec4 fragmentColor;
void main() { fragmentColor=vec4(vertexColor,1.0); }
)";
        if (!program_->addShaderFromSourceCode(QOpenGLShader::Vertex, vertex)
            || !program_->addShaderFromSourceCode(QOpenGLShader::Fragment, fragment)
            || !program_->link()) { error_ = program_->log(); return; }
        const float vertices[] = {
            -0.7f,-0.6f, 1.f,0.f,0.f,
             0.7f,-0.6f, 0.f,1.f,0.f,
             0.0f, 0.7f, 0.f,0.f,1.f
        };
        if (!vao_.create() || !vbo_.create()) { error_ = "VAO/VBO creation failed"; return; }
        QOpenGLVertexArrayObject::Binder binding(&vao_);
        if (!vbo_.bind() || !program_->bind()) { error_ = "GPU resource bind failed"; return; }
        vbo_.setUsagePattern(QOpenGLBuffer::StaticDraw);
        vbo_.allocate(vertices, sizeof(vertices));
        program_->enableAttributeArray(0);
        program_->setAttributeBuffer(0, GL_FLOAT, 0, 2, 5 * sizeof(float));
        program_->enableAttributeArray(1);
        program_->setAttributeBuffer(1, GL_FLOAT, 2 * sizeof(float), 3, 5 * sizeof(float));
        vbo_.release(); program_->release();
        ready_ = true;
    }
    void resizeGL(int, int) override {
        // QOpenGLWidget 已维护 framebuffer 和 viewport；矩阵扩展在教程中讲解。
    }
    void paintGL() override {
        if (!isValid() || !functionsReady_) return;
        glClearColor(0.06f, 0.10f, 0.16f, 1.f);
        glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT);
        if (!ready_) return;
        program_->bind();
        QOpenGLVertexArrayObject::Binder binding(&vao_);
        glDrawArrays(GL_TRIANGLES, 0, 3);
        program_->release();
    }
private:
    void cleanup() {
        if (!context() || !context()->isValid()) return;
        makeCurrent();
        ready_ = false;
        vao_.destroy(); vbo_.destroy(); program_.reset();
        doneCurrent();
    }
    QOpenGLBuffer vbo_{QOpenGLBuffer::VertexBuffer};
    QOpenGLVertexArrayObject vao_;
    std::unique_ptr<QOpenGLShaderProgram> program_;
    bool ready_ = false;
    bool functionsReady_ = false;
    QMetaObject::Connection cleanupConnection_;
    QString error_;
};

int main(int argc, char **argv) {
    QSurfaceFormat format;
    format.setRenderableType(QSurfaceFormat::OpenGL);
    format.setVersion(3, 3);
    format.setProfile(QSurfaceFormat::CoreProfile);
    format.setDepthBufferSize(24);
    QSurfaceFormat::setDefaultFormat(format);
    QApplication app(argc, argv);
    Triangle window;
    window.setWindowTitle(QStringLiteral("示例 17 · OpenGL 彩色三角形"));
    window.resize(640, 480); window.show();
    if (app.arguments().contains("--self-test")) {
        QTimer::singleShot(1500, &window, [&] {
            if (!window.ready()) { qCritical() << window.error(); app.exit(1); return; }
            const QImage image = window.grabFramebuffer();
            if (image.isNull()) { app.exit(2); return; }
            const auto center = image.pixelColor(image.width()/2, image.height()/2);
            const auto corner = image.pixelColor(4,4);
            if (center == corner) { qCritical() << "Triangle pixels missing"; app.exit(3); return; }
            qInfo() << "PASS OpenGL context, shaders, buffer, draw and pixel readback";
            app.exit(0);
        });
    }
    return app.exec();
}
