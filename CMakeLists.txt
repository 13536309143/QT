cmake_minimum_required(VERSION 3.21)
project(QtLearning LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTOUIC ON)
set(CMAKE_AUTORCC ON)
if(MSVC)
  add_compile_options(/utf-8 /W4)
endif()
option(COURSE_BUILD_QUICK "Build Qt Quick example" ON)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Network Sql Test)
set(QT_VERSION_MAJOR 6)
enable_testing()
function(course_example name)
  add_executable(${name} examples/${name}/main.cpp ${ARGN})
  target_link_libraries(${name} PRIVATE Qt${QT_VERSION_MAJOR}::Widgets Qt${QT_VERSION_MAJOR}::Network Qt${QT_VERSION_MAJOR}::Sql)
endfunction()
course_example(01_hello)
course_example(02_counter)
course_example(03_form)
course_example(04_editor)
course_example(05_json)
course_example(06_model)
course_example(07_paint)
course_example(08_worker)
course_example(09_http)
course_example(10_sql)
course_example(11_designer examples/11_designer/form.ui)
course_example(12_resources examples/12_resources/assets.qrc)
add_subdirectory(projects/taskdesk)
add_executable(course_tests tests/tst_course.cpp)
target_include_directories(course_tests PRIVATE projects/taskdesk)
target_link_libraries(course_tests PRIVATE Qt${QT_VERSION_MAJOR}::Test Qt${QT_VERSION_MAJOR}::Sql Qt${QT_VERSION_MAJOR}::Widgets)
add_test(NAME course_tests COMMAND course_tests -o -,txt -o ${CMAKE_CURRENT_BINARY_DIR}/qt-test-results.xml,junitxml)
set_tests_properties(course_tests PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
if(COURSE_BUILD_QUICK)
  add_subdirectory(examples/13_quick)
endif()
